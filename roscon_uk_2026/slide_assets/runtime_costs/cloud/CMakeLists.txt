cmake_minimum_required(VERSION 3.18)
project(cloud_comparison LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(nanobind CONFIG REQUIRED)
add_library(cloud_kernel SHARED kernel.cpp)
target_compile_features(cloud_kernel PRIVATE cxx_std_17)
pybind11_add_module(cloud_pybind11 bindings.cpp)
target_link_libraries(cloud_pybind11 PRIVATE cloud_kernel)
nanobind_add_module(cloud_nanobind bindings.cpp)
target_compile_definitions(cloud_nanobind PRIVATE CLOUD_NANOBIND)
target_link_libraries(cloud_nanobind PRIVATE cloud_kernel)
set_target_properties(cloud_pybind11 cloud_nanobind PROPERTIES BUILD_RPATH "$ORIGIN")
