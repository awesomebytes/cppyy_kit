cmake_minimum_required(VERSION 3.18)
project(eigen_step_native LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
add_library(eigen_step_core OBJECT ../../eigen_startup/step.cpp)
target_link_libraries(eigen_step_core PRIVATE Eigen3::Eigen)
target_compile_features(eigen_step_core PRIVATE cxx_std_17)
set_target_properties(eigen_step_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
pybind11_add_module(eigen_step_native bindings.cpp $<TARGET_OBJECTS:eigen_step_core>)
find_package(nanobind CONFIG REQUIRED)
nanobind_add_module(eigen_step_nanobind nanobind.cpp $<TARGET_OBJECTS:eigen_step_core>)
